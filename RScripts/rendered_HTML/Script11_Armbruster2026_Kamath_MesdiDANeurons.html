Script11_Armbruster2026_Kamath_MesdiDANeurons

Daniel Armbruster

2026-07-24

  • In this script we downloaded and reanalyzed the scRNA-seq data on the human substanitia nigra pars compacta (A9) DA neurons published by Kamath and colleagues (Kamath et al., 2022; DOI: 10.1038/s41593-022-01061-1)

  • The data is deposited on the single cell portal of the broad institute and can be downloaded from: https://singlecell.broadinstitute.org/single_cell/study/SCP1768/single-cell-genomic-profiling-of-human-dopamine-neurons-identifies-a-population-that-selectively-degenerates-in-parkinsons-disease-single-nuclei-data#study-download

  • The authors deposited the barcode, feature and matrix files as well as the final Seurat object metadata (containing cell type annotations) and UMAP coordinates of the relevant DA neurons. Thus, we need to create a Seurat object from the barcode, feature and matrix file, subset for the DA cell type containing clusters the authors present in Figure 1d and add their respective metadata and UMAP coordinates.

  • The session info including the used packages and their versions is provided at the end of this script.

seurat_object_to_analyse <- kamath_seurat

# Count threshold for the marker genes
expression_threshold <- 0
# Cell type cluster annotation
cell_clusters <- as.data.frame(seurat_object_to_analyse$Cell_Type)

# Read the neurotransmitter/neuropeptide file (production and reception markers)
gene_lists <- read.csv(file = "Resources/neurotransmitter_marker_human.csv", sep=";", skipNul = T, check.names = F)

# Get Expression Matrix from your Seurat object
expr_matrix <- GetAssayData(seurat_object_to_analyse, assay = "RNA", slot = "count")

# Get raw count expression matrix for the peptide/transmitter marker genes per cell and add cell type information
all_marker_genes <- unname(unlist(gene_lists))

valid_genes <- intersect(all_marker_genes, rownames(expr_matrix))
expr_matrix_relevant_genes <- expr_matrix[valid_genes, ]

expr_matrix_relevant_genes_clusters <- AverageExpression(

  slot     = "data",
  features = valid_genes,
  group.by = "Cell_Type",
  verbose  = FALSE
)

## Names of identity class contain underscores ('_'), replacing with dashes ('-')
## This message is displayed once per session.
expr_matrix_relevant_genes_clusters <- as.data.frame(expr_matrix_relevant_genes_clusters)

expr_matrix_relevant_genes_clusters <- expr_matrix_relevant_genes_clusters[match(valid_genes, rownames(expr_matrix_relevant_genes_clusters)), , drop = FALSE]

expr_matrix_relevant_genes_clusters <- t(expr_matrix_relevant_genes_clusters)




# Define all neurotransmitters
all_neurotransmitters <- unique(sub("_out$|_in$", "", colnames(gene_lists)))

# Initialize empty matrices for production and reception
production_matrix <- matrix(0, nrow = ncol(expr_matrix), ncol = length(all_neurotransmitters))
reception_matrix <- matrix(0, nrow = ncol(expr_matrix), ncol = length(all_neurotransmitters))

# Set cell names as row names for both matrices
rownames(production_matrix) <- colnames(expr_matrix)
rownames(reception_matrix) <- colnames(expr_matrix)

# Set neurotransmitters as column names
colnames(production_matrix) <- all_neurotransmitters
colnames(reception_matrix) <- all_neurotransmitters

# Track processed neurotransmitters
processed_neurotransmitters <- c()

# Loop through each neurotransmitter to process production and reception markers
for (gene_group_name in all_neurotransmitters) {

  # Identify production markers
  production_column <- paste0(gene_group_name, "_out")
  if (production_column %in% colnames(gene_lists)) {
    production_markers <- gene_lists[[production_column]][gene_lists[[production_column]] %in% rownames(expr_matrix)]
  } else {
    production_markers <- NULL
  }

  # Identify reception markers
  reception_column <- paste0(gene_group_name, "_in")
  if (reception_column %in% colnames(gene_lists)) {
    reception_markers <- gene_lists[[reception_column]][gene_lists[[reception_column]] %in% rownames(expr_matrix)]
  } else {
    reception_markers <- NULL
  }

  # Identify producing cells
  if (!is.null(production_markers) && length(production_markers) > 0) {
    production_cells <- colSums(expr_matrix[production_markers, , drop = FALSE] > expression_threshold) > 0
  } else {
    production_cells <- rep(FALSE, ncol(expr_matrix))  # Ensure 0 for all cells
  }
  
  # Identify receiving cells
  if (!is.null(reception_markers) && length(reception_markers) > 0) {
    reception_cells <- colSums(expr_matrix[reception_markers, , drop = FALSE] > expression_threshold) > 0
  } else {
    reception_cells <- rep(FALSE, ncol(expr_matrix))  # Ensure 0 for all cells
  }
  # Identify dbh_cells
  dbh_cells <- expr_matrix["DBH", ] > expression_threshold
  
  # Handle special cases for dopaminergic and noradrenergic cells
  if (gene_group_name == "dopa") {
    production_cells <- production_cells & !dbh_cells
    } else if (gene_group_name == "nora") {
      dbh_cells <- expr_matrix["DBH", ] > expression_threshold
      production_cells <- dbh_cells & production_cells
      }

  # Update production and reception matrices
  production_matrix[, gene_group_name] <- as.numeric(production_cells)
  reception_matrix[, gene_group_name] <- as.numeric(reception_cells)

  # Store the classifications in Seurat object metadata
  seurat_object_to_analyse[[paste0(gene_group_name, "_production")]] <- as.numeric(production_cells)
  seurat_object_to_analyse[[paste0(gene_group_name, "_reception")]] <- as.numeric(reception_cells)

  # Mark this neurotransmitter as processed
  processed_neurotransmitters <- c(processed_neurotransmitters, gene_group_name)
}

# Ensure all neurotransmitters are included in metadata, even if missing markers
for (gene_group_name in all_neurotransmitters) {
  if (!(gene_group_name %in% processed_neurotransmitters)) {
    seurat_object_to_analyse[[paste0(gene_group_name, "_production")]] <- 0
    seurat_object_to_analyse[[paste0(gene_group_name, "_reception")]] <- 0
    production_matrix[, gene_group_name] <- 0
    reception_matrix[, gene_group_name] <- 0
  }
}

# Threshold (% of cells that need to express neurotrans/neuropep) to get in plot 
threshold <- 0.1

# Get cluster identities from your Seurat object to include in the plot
clusters <- rev(levels(seurat_object_to_analyse@active.ident))

clusters <- str_sort(clusters, decreasing = T)

# Create an empty matrix for storing the Fractions for both production and reception
genegroup_Fraction_matrix <- matrix(0, nrow = length(clusters), ncol = length(colnames(gene_lists)))
rownames(genegroup_Fraction_matrix) <- clusters

# The column names will now include both 'production' and 'reception' for each neurotransmitter
col_names <- colnames(gene_lists)
col_names <- str_replace(col_names, "_out", "_production")
col_names <- str_replace(col_names, "_in", "_reception")
colnames(genegroup_Fraction_matrix) <- col_names

# Loop through each cluster and calculate the Fraction for each neurotransmitter
for (cluster in clusters) {
  # Filter the Seurat object by cluster
  cluster_cells <- WhichCells(seurat_object_to_analyse, idents = cluster)
  cluster
  for (name in colnames(gene_lists)) {
    # Strip the "_out" for production and "_in" for reception when accessing metadata
    name <- sub("_out$|_in$", "", name)
    
    # Construct the production and reception column names
    production_colname <- paste0(name, "_production")
    reception_colname <- paste0(name, "_reception")

    # Try to extract production data, otherwise return a vector of zeros
    production_cells <- try(seurat_object_to_analyse@meta.data[cluster_cells, production_colname], silent = TRUE)
    if (inherits(production_cells, "try-error")) {
      production_cells <- rep(0, length(cluster_cells))
    }

    # Try to extract reception data, otherwise return a vector of zeros
    reception_cells <- try(seurat_object_to_analyse@meta.data[cluster_cells, reception_colname], silent = TRUE)
    if (inherits(reception_cells, "try-error")) {
      reception_cells <- rep(0, length(cluster_cells))
    }

    # Calculate the Fractions
    production_Fraction <- sum(production_cells) / length(cluster_cells)
    reception_Fraction <- sum(reception_cells) / length(cluster_cells)

    # Store the Fraction values (try-catch ensures no indexing errors)
    try(genegroup_Fraction_matrix[cluster, paste(name, "production", sep = "_")] <- production_Fraction, silent = TRUE)
    try(genegroup_Fraction_matrix[cluster, paste(name, "reception", sep = "_")] <- reception_Fraction, silent = TRUE)
  }
}

    production_Fraction <- sum(production_cells) / length(cluster_cells)
    reception_Fraction <- sum(reception_cells) / length(cluster_cells)

    # Store the Fraction values (try-catch ensures no indexing errors)
    try(genegroup_Fraction_matrix[cluster, paste(name, "production", sep = "_")] <- production_Fraction, silent = TRUE)
    try(genegroup_Fraction_matrix[cluster, paste(name, "reception", sep = "_")] <- reception_Fraction, silent = TRUE)
  }
}
