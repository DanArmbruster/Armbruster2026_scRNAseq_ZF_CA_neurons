  • This script performed quality control, preprocessing, and merging of the published telencephalon 10X-based scRNAseq dataset from Pandey et al. 2023 (10.1101/gr.277278.122) with subsets of the annotated CA dataset we generated in Script 3.

  • The published whole telencephalon dataset consists of samples derived from 6 dpf, 15 dpf and adult zebrafish and was deposited as mapped barcode, features and matrix files in the NCBI Gene Expression Omnibus under the accession number GSE212314.

  • The session info including the used packages and their versions is provided at the end of this script.

3 Quality Control

3.1 Prepare Metadata for QC

  • Store genes detected per UMI, mitochondrial counts, cell names, and sample name in metadata
  • Rename count and feature columns
for (seurat_object in names(pandey_seurat_objects)) {
  
  # Extract seurat object to modify from list
  obj <- pandey_seurat_objects[[seurat_object]]
  
  # Calculate log10-transformed genes per UMI ratio
  obj$log10GenesPerUMI <- log10(obj$nFeature_RNA)/log10(obj$nCount_RNA)
  
  # Add mitochondrial gene expression percentage
  obj$MTratio <- PercentageFeatureSet(obj, pattern = "mt-*")
  
  # Add Cell name
  obj$cells <- rownames(obj@meta.data)
  obj@meta.data <- obj@meta.data %>%
    dplyr::rename(nUMI = nCount_RNA, nGene = nFeature_RNA)
  
  # Add sample ID
  obj$sample <- seurat_object
  
  #Assign modified seurat object back to list 
  pandey_seurat_objects[[seurat_object]] <- obj
  
}
  • Extract metadata from each Seurat object and generate a corresponding list for downstream QC assessment
pandey_metadata <- list()

for (seurat_object in names(pandey_seurat_objects)) {
  metadata <- pandey_seurat_objects[[seurat_object]]@meta.data
  pandey_metadata[[seurat_object]] <- metadata
}

6.1 Cell level filtering

    • Mtratio < 8 (only cells with less then 8 % mitochondrial reads)

5.2 Gene level filtering

  • Filter out genes with 0 counts over all cells and genes that are not detected in at least 3 cells.
# Only keep genes expressed in at least 3 cells
merged_seurat_object_filtered <- JoinLayers(merged_seurat_object_filtered) 
allown_counts <- GetAssayData(object = merged_seurat_object_filtered, layer = "counts")
allown_nonzero <- allown_counts > 0
allown_keep_genes <- Matrix::rowSums(allown_nonzero) >=3
allown_filtered_counts <- allown_counts[allown_keep_genes,]
merged_seurat_object_filtered <- CreateSeuratObject(allown_filtered_counts, meta.data = merged_seurat_object_filtered@meta.data)
